def digest(data): return hashlib.sha256(data).hexdigest()
def json_file(path): return json.loads(path.read_text())
def exit_file(path):

def head(root):

def go_counts(path):
    terminals,outputs,packages,parse_failures={},{},{},[]
    for number,raw in enumerate(path.read_text().splitlines(),1):
        if not raw.strip(): continue
        try: event=json.loads(raw)
        except json.JSONDecodeError: parse_failures.append(number); continue
        action,package,test=event.get("Action"),event.get("Package",""),event.get("Test")
        if test and action=="output": outputs.setdefault((package,test),[]).append(event.get("Output","").strip())
        if test and action in {"pass","fail","skip"}: terminals[(package,test)]=action
        elif not test and action in {"pass","fail","skip"}: packages[package]=action
    tests=sorted(terminals)
    leaves=[(p,t) for p,t in tests if not any(op==p and ot.startswith(t+"/") for op,ot in tests)]
    skips=[]
    for item in leaves:
        if terminals[item]=="skip":
            lines=outputs.get(item,[])
            reason=next((x for x in lines if "NOT_RUN" in x),next((x for x in lines if "SKIP" in x),""))
            skips.append({"package":item[0],"test":item[1],"reason":reason})
    return {"terminal":dict(Counter(terminals.values())),"leaf":dict(Counter(terminals[x] for x in leaves)),"packages":dict(Counter(packages.values())),"leaf_skips":skips,"parse_failures":parse_failures}
def normalized_counts(data): return {k:data.get(k) for k in ("terminal","leaf","packages","leaf_skips","parse_failures")}
def ledger(path):
    result={}
    for line in path.read_text().splitlines():
        m=re.fullmatch(r"(\d{4}) ([0-9a-f]{64})",line.strip())
        if not m: continue
        if m.group(1) in result: raise ValueError("duplicate ledger version")
        result[m.group(1)]=m.group(2)
    if not result: raise ValueError("missing ledger")
    return result

def known_secrets(payload=None):
    if payload is not None:
        return int(payload["private_source_count"]), sorted(set(payload["known_secrets"]), key=len, reverse=True)
    files=sorted({*Path("/private/tmp").glob("a03-task12*/**/*.env"),*Path("/private/tmp").glob("a03-task12*/**/*env.sh"),*Path("/private/tmp").glob("a03-task12*/**/*credentials*")})
    names=re.compile(r"(?:PASS(?:WORD)?|SECRET|TOKEN|HMAC|API_KEY|MYSQL_PWD|CREDENTIAL)",re.I)
    known=set()
    for path in files:
        try: text=path.read_text(errors="replace")
        except OSError: continue
        for line in text.splitlines():
            line=line.strip()[7:] if line.strip().startswith("export ") else line.strip()
            if "=" not in line: continue
            name,raw=line.split("=",1); value=raw.strip()
            try:
                parsed=shlex.split(value,posix=True)
                if len(parsed)==1: value=parsed[0]
            except ValueError: pass
            if names.search(name) and len(value)>=8 and value.lower() not in {"<redacted>","redacted","false","true","null","none"}: known.add(value)
            if name.upper().endswith("DATABASE_URL") or ("MYSQL" in name.upper() and "://" in value):
                try:
                    password=urlparse(value).password
                    if password and len(unquote(password))>=8: known.add(unquote(password))
                except ValueError: pass
    return len(files),sorted(known,key=len,reverse=True)
