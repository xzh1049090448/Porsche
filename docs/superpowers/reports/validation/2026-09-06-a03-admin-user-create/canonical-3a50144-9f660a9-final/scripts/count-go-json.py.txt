#!/usr/bin/env python3
import json
import pathlib
import sys

source = pathlib.Path(sys.argv[1])
destination = pathlib.Path(sys.argv[2])
terminals = {}
outputs = {}
packages = {}
fail_events = []

for number, raw in enumerate(source.read_text(encoding="utf-8").splitlines(), 1):
    if not raw.strip():
        continue
    try:
        event = json.loads(raw)
    except json.JSONDecodeError as exc:
        fail_events.append({"line": number, "error": f"invalid_json:{exc.msg}"})
        continue
    action = event.get("Action")
    package = event.get("Package", "")
    test = event.get("Test")
    if test and action == "output":
        outputs.setdefault((package, test), []).append(event.get("Output", "").strip())
    if test and action in {"pass", "fail", "skip"}:
        terminals[(package, test)] = action
    elif not test and action in {"pass", "fail", "skip"}:
        packages[package] = action

tests = sorted(terminals)
leaves = []
for package, test in tests:
    prefix = test + "/"
    if not any(other_package == package and other_test.startswith(prefix) for other_package, other_test in tests):
        leaves.append((package, test))

def counts(items):
    result = {}
    for action in items:
        result[action] = result.get(action, 0) + 1
    return result

leaf_skips = []
for package, test in leaves:
    if terminals[(package, test)] != "skip":
        continue
    lines = outputs.get((package, test), [])
    reason = next((line for line in lines if "NOT_RUN" in line), "")
    if not reason:
        reason = next((line for line in lines if "SKIP" in line), "")
    leaf_skips.append({"package": package, "test": test, "reason": reason})

result = {
    "source": source.name,
    "terminal": counts(terminals.values()),
    "leaf": counts(terminals[item] for item in leaves),
    "packages": counts(packages.values()),
    "leaf_skips": leaf_skips,
    "parse_failures": fail_events,
}
destination.write_text(json.dumps(result, indent=2, sort_keys=True) + "\n", encoding="utf-8")
print(json.dumps(result, sort_keys=True))
