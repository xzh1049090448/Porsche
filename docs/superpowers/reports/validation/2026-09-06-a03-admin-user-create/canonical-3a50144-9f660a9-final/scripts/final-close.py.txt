def run(argv, check=True, cwd=BE):
    result = subprocess.run(argv, cwd=cwd, text=True, errors="replace", capture_output=True)
    if check and result.returncode:
        raise RuntimeError("command failed: " + " ".join(argv[:3]))
    return result


def docker_inspect(ids):
    if not ids:
        return []
    return json.loads(run(["docker", "inspect", *ids]).stdout)

def container_snapshot(inspect):
    return {
        "id": inspect["Id"],
        "name": inspect["Name"].lstrip("/"),
        "configured_image": inspect["Config"]["Image"],
        "image_id": inspect["Image"],
        "labels": inspect["Config"].get("Labels") or {},
        "ports": inspect["NetworkSettings"].get("Ports") or {},
        "mounts": inspect.get("Mounts") or [],
        "tmpfs": inspect["HostConfig"].get("Tmpfs") or {},
        "auto_remove": bool(inspect["HostConfig"].get("AutoRemove")),
        "state": inspect["State"]["Status"],
    }


def all_containers():
    ids = run(["docker", "ps", "-aq"]).stdout.split()
    return [container_snapshot(item) for item in docker_inspect(ids)]


def image_inventory():
    result = run(["docker", "images", "--no-trunc", "--digests", "--format", "{{.ID}}\t{{.Repository}}\t{{.Tag}}\t{{.Digest}}"])
    return sorted(line for line in result.stdout.splitlines() if line)


def volume_inventory():
    names = run(["docker", "volume", "ls", "-q"]).stdout.split()
    if not names:
        return []
    rows = json.loads(run(["docker", "volume", "inspect", *names]).stdout)
    return sorted(
        [
            {"name": row["Name"], "driver": row["Driver"], "labels": row.get("Labels") or {}, "options": row.get("Options") or {}, "scope": row.get("Scope")}
            for row in rows
        ],
        key=lambda row: row["name"],
    )


def private_paths():

def collect_known():
    files = sorted({
        *Path("/private/tmp").glob("a03-task12*/**/*.env"),
        *Path("/private/tmp").glob("a03-task12*/**/*env.sh"),
        *Path("/private/tmp").glob("a03-task12*/**/*credentials*"),
    })
    names = re.compile(r"(?:PASS(?:WORD)?|SECRET|TOKEN|HMAC|API_KEY|MYSQL_PWD|CREDENTIAL)", re.I)
    placeholder = re.compile(r"^(?:<redacted>|redacted|false|true|null|none)?$", re.I)
    known = set()
    for path in files:
        text = path.read_text(errors="replace")
        for line in text.splitlines():
            line = line.strip()[7:] if line.strip().startswith("export ") else line.strip()
            if "=" not in line:
                continue
            name, raw = line.split("=", 1)
            value = raw.strip()
            try:
                parsed = shlex.split(value, posix=True)
                if len(parsed) == 1:
                    value = parsed[0]
            except ValueError:
                pass
            if names.search(name) and len(value) >= 8 and not placeholder.fullmatch(value):
                known.add(value)
            if name.upper().endswith("DATABASE_URL") or ("MYSQL" in name.upper() and "://" in value):
                try:
                    password = urlparse(value).password
                    if password and len(unquote(password)) >= 8:
                        known.add(unquote(password))
                except ValueError:
                    pass
    if len(files) != 19 or len(known) != 34:
        raise RuntimeError("known-secret provenance count mismatch")
    return files, sorted(known, key=len, reverse=True)


def app_state():
    listener_checks = {}

    process = run(["ps", "-p", ",".join(str(pid) for pid in APP_PIDS), "-o", "pid=,ppid=,state=,etime=,command="], check=False)
    return listener_checks, {"exit": process.returncode, "stdout_empty": not bool(process.stdout)}
