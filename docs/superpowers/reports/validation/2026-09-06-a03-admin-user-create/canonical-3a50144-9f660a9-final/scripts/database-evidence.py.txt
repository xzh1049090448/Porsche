def run_sql(statement):
    env = os.environ.copy()
    env["MYSQL_PWD"] = url.password or ""
    result = subprocess.run(
        ["docker", "exec", "-i", "-e", "MYSQL_PWD", container, "mysql",
         "-h127.0.0.1", "-u" + (url.username or ""), "-N", "-B", database],
        input=statement.rstrip(";\n") + ";\n", env=env, text=True,
        capture_output=True,
    )
    if result.returncode:
        safe_error = (result.stderr or "").replace(url.password or "", "<redacted>").strip()
        raise RuntimeError("disposable database query failed: " + safe_error[:500])
    return result.stdout.rstrip("\n")

def rows(statement):
    out = run_sql(statement)
    return [] if not out else [line.split("\t") for line in out.splitlines()]

def integer(statement):
    out = run_sql(statement).strip()
    if not re.fullmatch(r"-?\d+", out):
        raise RuntimeError("unexpected scalar query result")
    return int(out)

def q(value):
    return "'" + value.replace("\\", "\\\\").replace("'", "''") + "'"

def enum(value, mapping):
    return mapping.get(int(value), "unknown") if value != "NULL" else None
