import argparse
import json
import os
import re
import shlex
import subprocess
import sys
from collections import Counter
from pathlib import Path
from urllib.parse import unquote, urlparse

base = Path("/Users/xuzhihao/code/Porsche/.worktrees/admin-public-260903/docs/superpowers/reports/validation/2026-09-06-a03-admin-user-create")
canonical = base / "canonical-3a50144-9f660a9-final"

parser = argparse.ArgumentParser()
parser.add_argument("--payload-stdin", action="store_true")
arguments = parser.parse_args()

private_files = sorted({
    *Path("/private/tmp").glob("a03-task12*/**/*.env"),
    *Path("/private/tmp").glob("a03-task12*/**/*env.sh"),
    *Path("/private/tmp").glob("a03-task12*/**/*credentials*"),
})
secret_name = re.compile(r"(?:PASS(?:WORD)?|SECRET|TOKEN|HMAC|API_KEY|MYSQL_PWD|CREDENTIAL)", re.I)
placeholder = re.compile(r"^(?:<redacted>|redacted|false|true|null|none)?$", re.I)
known = set()
private_source_count = len(private_files)
payload_container_logs = None

def decoded_assignment(raw):
    value = raw.strip()
    try:
        parsed = shlex.split(value, posix=True)
        if len(parsed) == 1:
            value = parsed[0]
    except ValueError:
        pass
    return value

if arguments.payload_stdin:
    payload = json.load(sys.stdin)
    private_source_count = int(payload["private_source_count"])
    known.update(payload["known_secrets"])
    payload_container_logs = list(payload["container_logs"])
else:
    for path in private_files:
        try:
            text = path.read_text(errors="replace")
        except OSError:
            continue
        for line in text.splitlines():
            line = line.strip()
            if line.startswith("export "):
                line = line[7:]
            if "=" not in line:
                continue
            name, raw = line.split("=", 1)
            value = decoded_assignment(raw)
            if secret_name.search(name) and len(value) >= 8 and not placeholder.fullmatch(value):
                known.add(value)
            if name.strip().upper().endswith("DATABASE_URL") or "MYSQL" in name.upper() and "://" in value:
                try:
                    parsed = urlparse(value)
                    if parsed.password and len(unquote(parsed.password)) >= 8:
                        known.add(unquote(parsed.password))
                except ValueError:
                    pass

patterns = {
    "opaque_ticket_or_idempotency": re.compile(r"(?<![A-Za-z0-9_-])(?:av|ik)_[A-Za-z0-9_-]{43}(?![A-Za-z0-9_-])"),
    "bearer_value": re.compile(r"(?i)\bBearer\s+[A-Za-z0-9._~+/-]{16,}"),
    "json_credential_literal": re.compile(r'''(?i)["'](?:password|current_password|ticket|idempotency[_-]?key|authorization|jwt_secret(?:_key)?|auth_hmac_key|action_security_hmac_key|admin_token|metrics_token|api_key)["']\s*:\s*["'](?!<redacted>)[^"'\r\n]{6,}["']'''),
    "shell_credential_literal": re.compile(r'''(?im)^\s*(?:export\s+)?[A-Za-z0-9_]*(?:PASSWORD|SECRET|HMAC_KEY|ADMIN_TOKEN|METRICS_TOKEN|API_KEY|MYSQL_PWD)[A-Za-z0-9_]*\s*=\s*(?!<redacted>|\$|process\.|os\.environ|strings\.|decoded|value|raw)["']?[^\s"']{8,}'''),
    "private_key_material": re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----"),
}

def category(relative):
    value = relative.as_posix()
    if "/scripts/" in "/" + value:
        return "reproduction_source_reference"
    name = relative.name
    if re.search(r"(?:backend-(?:focused|full|race)|frontend-test)", name) and relative.suffix.lower() in {".json", ".jsonl", ".tap", ".log"}:
        return "test_output_fixture_reference"
    return "runtime_or_evidence"

totals = Counter()
pattern_counts = {name: Counter() for name in patterns}
text_files = 0
binary_files = 0
text_bytes = 0
exact_files = 0
excluded_self_outputs = {
    canonical / "secret-scan.json",
    canonical / "manifest.json",
}
all_artifact_files = sorted(file for file in base.rglob("*") if file.is_file())
for path in all_artifact_files:
    if path in excluded_self_outputs:
        continue
    data = path.read_bytes()
    if path.suffix.lower() in {".png", ".jpg", ".jpeg", ".webp", ".gif", ".zip"} or b"\0" in data[:4096]:
        binary_files += 1
        continue
    text = data.decode("utf-8", errors="replace")
    text_files += 1
    text_bytes += len(data)
    group = category(path.relative_to(base))
    exact_here = sum(text.count(secret) for secret in known)
    totals["exact_known_secret_matches"] += exact_here
    if exact_here:
        exact_files += 1
    for name, pattern in patterns.items():
        count = len(pattern.findall(text))
        pattern_counts[name][group] += count

docker_logs = []
if payload_container_logs is not None:
    docker_logs = payload_container_logs
else:
    for container in ["a03-task12-mysql-20260907182125-bfacc639", "a03-task12-redis-20260907182125-bfacc639"]:
        result = subprocess.run(["docker", "logs", container], text=True, capture_output=True)
        if result.returncode:
            raise RuntimeError("task container log unavailable")
        docker_logs.append((result.stdout or "") + (result.stderr or ""))
docker_text = "\n".join(docker_logs)
docker_exact = sum(docker_text.count(secret) for secret in known)
docker_patterns = {name: len(pattern.findall(docker_text)) for name, pattern in patterns.items()}

browser = json.loads((canonical / "browser/browser-acceptance.json").read_text())
browser_runtime_scan = browser["runtime_console_secret_scan"]
browser_console = {
    "debug": int(browser["console_counts"].get("debug", 0)),
    "error": int(browser["console_counts"].get("error", 0)),
    "expected_exercised_http_failure": int(browser["console_classifications"].get("expected_exercised_http_failure", 0)),
    "page_error": int(browser["page_error_count"]),
    "stored_message_values": 0,
}

runtime_pattern_total = sum(counts["runtime_or_evidence"] for counts in pattern_counts.values())
docker_pattern_total = sum(docker_patterns.values())
status = "PASS" if (
    private_source_count == 19 and len(known) == 34 and len(docker_logs) == 8 and
    totals["exact_known_secret_matches"] == 0 and exact_files == 0 and runtime_pattern_total == 0 and
    docker_exact == 0 and docker_pattern_total == 0 and
    int(browser_runtime_scan["exact_known_secret_matches"]) == 0 and
    int(browser_runtime_scan["credential_pattern_matches"]) == 0 and browser_console["page_error"] == 0
) else "FAIL"

result = {
    "status": status,
    "scope": {
        "artifact_root": "all canonical and superseded text artifacts under A03 validation report",
        "artifact_files_observed": len(all_artifact_files),
        "text_files_scanned": text_files,
        "text_bytes_scanned": text_bytes,
        "binary_artifacts_skipped_as_nontext": binary_files,
        "private_secret_source_files_read_in_memory": private_source_count,
        "known_secret_values_scanned": len(known),
        "task_container_logs_scanned_in_memory": len(docker_logs),
        "self_outputs_excluded": ["canonical-3a50144-9f660a9-final/secret-scan.json", "canonical-3a50144-9f660a9-final/manifest.json"],
    },
    "exact_known_secret_scan": {
        "matches": int(totals["exact_known_secret_matches"]),
        "files_with_matches": exact_files,
        "stored_match_values": 0,
        "stored_match_lines": 0,
    },
    "credential_pattern_scan": {
        name: {
            "runtime_or_evidence": int(counts["runtime_or_evidence"]),
            "test_output_fixture_reference": int(counts["test_output_fixture_reference"]),
            "reproduction_source_reference": int(counts["reproduction_source_reference"]),
            "stored_match_values": 0,
            "stored_match_lines": 0,
        }
        for name, counts in pattern_counts.items()
    },
    "container_console_scan": {
        "exact_known_secret_matches": docker_exact,
        "credential_pattern_matches": docker_pattern_total,
        "stored_console_values": 0,
    },
    "browser_runtime_console_scan": {
        "exact_known_secret_matches": int(browser_runtime_scan["exact_known_secret_matches"]),
        "credential_pattern_matches": int(browser_runtime_scan["credential_pattern_matches"]),
        "console_classification_counts": browser_console,
    },
    "classification_policy": {
        "runtime_or_evidence": "must be zero for exact secrets and credential patterns",
        "test_output_fixture_reference": "pattern-only fixture literals are classified separately; exact task-secret matches still fail",
        "reproduction_source_reference": "pattern syntax and environment-variable references are classified separately; exact task-secret matches still fail",
    },
}
(canonical / "secret-scan.json").write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
print(f"secret_scan={status} text_files={text_files} known_values={len(known)} exact={totals['exact_known_secret_matches']} runtime_patterns={runtime_pattern_total} docker_patterns={docker_pattern_total}")
raise SystemExit(0 if status == "PASS" else 1)
